\subsection{Bonnes pratiques de programmation 2 : ne vous répétez pas !}
\begin{frame}[fragile]
\begin{lstlisting}[language={[ansi]C},title=Exemple de répétition : il faut factoriser le code de clignotement]
// ...
void loop() {
  // ...
  blinkRedLed();
  blinkBlueLed();
  // ...
}
void blinkRedLed() {
  compute = time % T_LED_RED;
  if(compute < T_LED_RED/2) digitalWrite(LED_RED_PIN, HIGH);
  else digitalWrite(LED_RED_PIN, LOW);
}
void blinkBlueLed() {
  compute = time % T_LED_BLUE;
  if(compute < T_LED_BLUE/2) digitalWrite(LED_BLUE_PIN, HIGH);
  else digitalWrite(LED_BLUE_PIN, LOW);
}
\end{lstlisting}
\end{frame}
\begin{frame}[fragile]
\begin{lstlisting}[language={[ansi]C},title=Exemple de répétition : factorisation effectuée]
// ...
void loop() {
  // ...
  blinkLed(LED_RED_PIN, T_LED_RED);
  blinkLed(LED_BLUE_PIN, T_LED_BLUE);
  // ...
}
void blinkLed(int ledPin, int period) {
  compute = time % period;
  if(compute < period/2) digitalWrite(ledPin, HIGH);
  else digitalWrite(ledPin, LOW);
}
\end{lstlisting}
\end{frame}

\begin{frame}{Appuie sur le bouton poussoir}
\begin{figure}
   \includegraphics[width=0.45\textwidth]{images/TEK0001.png}
\end{figure}
\begin{alertblock}{Mise en contact des lames métalliques :}
	\begin{itemize}
		\item Le doigt plaque et maintient les lames métalliques en contact 
		\item D'un point de vue mécanique, il n'y a pas de rebonds...
		\item ... ou en dessous du niveau logique haut
	\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{Relâchement du bouton poussoir}
\begin{figure}
   \includegraphics[width=0.45\textwidth]{images/TEK0000.png}
\end{figure}
\begin{alertblock}{Déconnexion des lames métalliques :}
	\begin{itemize}
		\item Le doigt libère une des lames métalliques
		\item D'un point de vue mécanique, les rebonds sont très probables
		\item Niveau logique bas retrouvé lors des rebonds
	\end{itemize}
\end{alertblock}
\end{frame}
